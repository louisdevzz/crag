\chapter{Mở rộng giao diện người dùng và quản lý bộ nhớ phiên}

\section{Kiến trúc tích hợp giao diện, cổng API và bộ nhớ}

Sau khi lõi tác tử và cơ chế CRAG đã được chứng minh hiệu quả độc lập qua các kịch bản kiểm chuẩn, hệ thống được mở rộng một giao diện người dùng hiện đại (Next.js 14) và cơ chế quản lý ngữ cảnh phiên hội thoại.

\begin{figure}[htbp]
    \centering
    \includegraphics[width=0.95\textwidth]{images/memory_architecture.png}
    \caption{Phân cấp ba tầng bộ nhớ (làm việc, lịch sử, ngữ nghĩa) và tường lửa cách ly đa người dùng.}
    \label{fig:memory_architecture}
\end{figure}

Giao diện Next.js lưu trữ một mã định danh ẩn danh (\texttt{client\_id}) trên trình duyệt thông qua \texttt{localStorage} và tự động sinh \texttt{session\_id} ngẫu nhiên cho từng phiên làm việc. Cổng API FastAPI tiếp nhận yêu cầu, phối hợp cùng Context Manager để nạp lịch sử tương tác gần nhất và thông tin hồ sơ doanh nghiệp đã xác thực, truyền vào \texttt{AgentState} nhưng \textbf{cô lập hoàn toàn khỏi danh mục trích dẫn pháp lý}.

\begin{calloutbox}{Nguyên tắc an toàn tối thượng về bộ nhớ}
Bộ nhớ chỉ đóng vai trò hỗ trợ giải thích bối cảnh và giải nghĩa đại từ. Mọi phán quyết và kết luận pháp lý đều bắt buộc phải được truy hồi lại từ các văn bản hiện hành trong cơ sở dữ liệu pháp luật hoặc cổng thông tin web chính thống. Tuyệt đối không sử dụng thông tin lưu trong bộ nhớ như một nguồn căn cứ pháp luật.
\end{calloutbox}

\section{Thiết kế cơ sở dữ liệu cho định danh, phiên và bộ nhớ}

Toàn bộ phân hệ người dùng và bộ nhớ được chuẩn hóa trong SQLite \texttt{app.db} với 4 bảng quan hệ ràng buộc khóa ngoại chặt chẽ:

\begin{lstlisting}[language=SQL, caption={Lược đồ quản lý định danh, phiên hội thoại và bộ nhớ người dùng}]
CREATE TABLE IF NOT EXISTS clients (
    id TEXT PRIMARY KEY,
    memory_enabled INTEGER DEFAULT 1,
    created_at TEXT DEFAULT CURRENT_TIMESTAMP,
    last_seen_at TEXT DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS sessions (
    id TEXT PRIMARY KEY,
    client_id TEXT NOT NULL,
    title TEXT,
    started_at TEXT DEFAULT CURRENT_TIMESTAMP,
    last_active_at TEXT DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(client_id) REFERENCES clients(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS messages (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    session_id TEXT NOT NULL,
    client_id TEXT NOT NULL,
    role TEXT NOT NULL, -- 'user' | 'assistant'
    content TEXT NOT NULL,
    route TEXT,
    crag_action TEXT,
    source_type TEXT,
    evidence_json TEXT,
    citation_report_json TEXT,
    created_at TEXT DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY(session_id) REFERENCES sessions(id) ON DELETE CASCADE,
    FOREIGN KEY(client_id) REFERENCES clients(id) ON DELETE CASCADE
);

CREATE TABLE IF NOT EXISTS memories (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    client_id TEXT NOT NULL,
    memory_key TEXT NOT NULL,
    memory_value TEXT NOT NULL,
    confidence REAL DEFAULT 1.0,
    updated_at TEXT DEFAULT CURRENT_TIMESTAMP,
    UNIQUE(client_id, memory_key),
    FOREIGN KEY(client_id) REFERENCES clients(id) ON DELETE CASCADE
);
\end{lstlisting}

\section{Kiểm soát bộ nhớ ngữ nghĩa qua danh mục cho phép}

Hệ thống áp dụng cơ chế trích xuất có kiểm soát nghiêm ngặt trong \texttt{memory/extractor.py} nhằm bảo vệ quyền riêng tư và tránh ô nhiễm bộ nhớ:

\begin{lstlisting}[language=Python, caption={Cơ chế kiểm soát danh mục cho phép và trích xuất hồ sơ}]
ALLOWED_MEMORY_KEYS = {
    "business_type",      # Loai hinh: Cong ty TNHH, Co phan, Doanh nghiep tu nhan
    "industry",           # Nganh nghe: Xay dung, Thuong mai dien tu, Cong nghe
    "province",           # Dia ban hoat dong: Ha Noi, TP.HCM, Long An, Da Nang
    "frequent_topic",     # Chu de quan tam: Lao dong, Hop dong, Giay phep
    "preferred_answer"    # Phong cach trinh bay: Ngan gon, Chi tiet
}

# Quy tac trich xuat ho so nguoi dung:
# 1. Chi trich xuat cac thuoc tinh nam trong danh muc ALLOWED_MEMORY_KEYS.
# 2. Tuyet doi khong luu mat khau, so CCCD/CMND, bi mat kinh doanh.
# 3. Gan diem tin cay (confidence) tu 0.0 den 1.0.
# 4. Neu nguoi dung phu nhan thong tin cu, tu dong cap nhat gia tri moi.
\end{lstlisting}

\section{Câu lệnh nhắc phân tách bối cảnh người dùng và căn cứ pháp lý}

\begin{lstlisting}[caption={Câu lệnh nhắc trả lời kết hợp bối cảnh có kiểm soát}]
Bối cảnh người dùng (Chỉ dùng để hiểu hoàn cảnh, không có giá trị pháp lý):
{memory_context}

Căn cứ pháp lý hiện tại (Nguồn căn cứ duy nhất để đưa ra kết luận):
{legal_context}

Câu hỏi hiện tại:
{query}

Quy tắc thực thi:
1. Bộ nhớ chỉ giúp giải thích đại từ (ví dụ: "công ty tôi" -> Công ty TNHH tại Long An).
2. Tuyệt đối không lấy thông tin từ câu trả lời cũ thay thế cho văn bản pháp luật hiện hành.
3. Mọi kết luận pháp lý đưa ra bắt buộc phải trích dẫn source_id từ danh mục căn cứ pháp lý.
4. Nếu căn cứ không có điều khoản tương ứng, nói rõ: "Chưa đủ căn cứ pháp lý để kết luận".
\end{lstlisting}

\section{Bảo mật và tiêu chuẩn cách ly dữ liệu đa người dùng}

\begin{keypoint}
\textbf{Tiêu chuẩn an toàn cách ly:} Hệ thống bảo đảm chỉ số an toàn \textbf{Cross-client Contamination = 0}. Trong toàn bộ các ca kiểm thử bảo mật, dữ liệu lịch sử hoặc bộ nhớ của khách hàng A tuyệt đối không bao giờ xuất hiện trong phiên truy vấn của khách hàng B. Mọi câu lệnh truy vấn cơ sở dữ liệu đều bắt buộc có ràng buộc tham số kép:
\begin{center}
\texttt{WHERE client\_id = ? AND session\_id = ?}
\end{center}
\end{keypoint}

\section{Đánh giá thử nghiệm A/B hiệu quả bộ nhớ}

Tiến hành đánh giá A/B trên 3 cấu hình thử nghiệm:
\begin{itemize}
    \item \textbf{Cấu hình A (Không bộ nhớ):} Tác tử xử lý độc lập từng câu hỏi thô riêng rẽ.
    \item \textbf{Cấu hình B (Lịch sử ngắn hạn):} Nạp 6 lượt tương tác gần nhất trong phiên hội thoại.
    \item \textbf{Cấu hình C (Lịch sử kết hợp hồ sơ ngữ nghĩa):} Kết hợp toàn diện lịch sử tương tác và thuộc tính doanh nghiệp.
\end{itemize}

Kết quả thực nghiệm cho thấy Cấu hình C nâng độ chính xác giải tham chiếu đại từ (\textit{Context Reuse Accuracy}) từ $12.5\%$ lên \textbf{$95.0\%$}, trong khi chỉ số \textit{Faithfulness} vẫn được bảo toàn trọn vẹn ở mức $1.0$ nhờ vào rào chắn an toàn bộ nhớ.
