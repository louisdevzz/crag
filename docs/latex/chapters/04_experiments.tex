\subsection{Các biến thể nghiên cứu tách biệt thành phần (Ablation Study)}
\begin{itemize}
    \item \textbf{Ablation 1 (Không tinh lọc):} Full CRAG nhưng loại bỏ bước bóc tách dải quy định (\textit{legal strips}), nạp nguyên khối văn bản dài vào LLM.
    \item \textbf{Ablation 2 (Không viết lại truy vấn):} Full CRAG nhưng loại bỏ bước viết lại truy vấn, dùng nguyên văn câu hỏi thô để tra cứu web.
    \item \textbf{Ablation 3 (Không lọc trang web):} Lấy toàn bộ văn bản thô từ kết quả tìm kiếm mà không qua bộ lọc dải quy định liên quan.
    \item \textbf{Ablation 4 (Chỉ truy hồi vector):} Thay thế truy hồi lai bằng truy hồi vector ChromaDB thuần túy (loại bỏ BM25 và RRF).
    \item \textbf{Ablation 5 (Không xếp hạng lại):} Bỏ mô hình Cross-Encoder, dùng trực tiếp độ tương đồng Cosine của vector để định tuyến và phân ngưỡng.
\end{itemize}

\section{Thiết kế tập hiệu chuẩn và tập kiểm thử độc lập}

\begin{calloutbox}{Quy tắc vàng trong đánh giá học thuật}
Tuyệt đối không sử dụng cùng một tập dữ liệu vừa để tìm ngưỡng $T_{low}/T_{high}$ vừa để báo cáo kết quả cuối cùng. Hệ thống phân tách độc lập hoàn toàn hai tập dữ liệu:
\end{calloutbox}

\begin{enumerate}
    \item \textbf{Tập hiệu chuẩn (Calibration Set, 20 câu):} Dành riêng cho thực nghiệm quét lưới (\textit{grid search}) tìm cặp ngưỡng $(T_{low}, T_{high})$ tối ưu.
    \item \textbf{Tập kiểm thử độc lập (Held-out Test Set, 40 câu):} Được khóa cố định trước khi chạy benchmark, không can thiệp vào quá trình hiệu chỉnh (Bảng~\ref{tab:test_set_distribution}).
\end{enumerate}

\section{Hệ thống chỉ số đánh giá toàn diện}

\subsection{Chỉ số đánh giá truy hồi}
\begin{itemize}
    \item \textbf{Recall@K:} Tỷ lệ bằng chứng pháp lý đúng xuất hiện trong Top-K tài liệu truy hồi.
    \item \textbf{MRR (Mean Reciprocal Rank):} Vị trí nghịch đảo trung bình của bằng chứng đúng đầu tiên:
    \begin{equation}
    \text{MRR} = \frac{1}{|Q|} \sum_{i=1}^{|Q|} \frac{1}{\text{rank}_i}
    \end{equation}
    \item \textbf{nDCG@K (Normalized Discounted Cumulative Gain):} Đánh giá chất lượng của toàn bộ thứ hạng sắp xếp.
\end{itemize}

\subsection{Chỉ số đánh giá bộ điều hướng và tìm kiếm mở rộng}
Nhằm lượng hóa mức độ chính xác khi kích hoạt tìm kiếm ngoài của tác tử:
\begin{align}
\text{Fallback Precision} &= \frac{\text{Số lần kích hoạt web đúng khi ngoài kho dữ liệu}}{\text{Tổng số lần hệ thống kích hoạt tìm kiếm web}} \\[0.5em]
\text{Fallback Recall} &= \frac{\text{Số câu hỏi ngoài kho dữ liệu được phát hiện thành công}}{\text{Tổng số câu hỏi ngoài kho dữ liệu trong tập kiểm thử}} \\[0.5em]
\text{False Fallback Rate} &= \frac{\text{Số câu hỏi có sẵn trong kho nhưng bị gọi nhầm ra web}}{\text{Tổng số câu hỏi có sẵn trong kho dữ liệu}}
\end{align}

\subsection{Chỉ số đánh giá chất lượng sinh và chuẩn mực pháp lý}
\begin{itemize}
    \item \textbf{RAGAS Faithfulness:} Đo lường tỷ lệ các mệnh đề kết luận trong câu trả lời có thể suy dẫn trực tiếp từ bằng chứng pháp lý (\textit{groundedness}).
    \item \textbf{RAGAS Answer Relevancy:} Mức độ bám sát câu hỏi người dùng thông qua độ tương đồng embedding với câu hỏi tái sinh.
    \item \textbf{RAGAS Context Precision \& Context Recall:} Đo lường tỷ lệ thông tin nhiễu trong bằng chứng và mức độ bao phủ sự thật khách quan.
    \item \textbf{Citation Accuracy:} Tỷ lệ trích dẫn pháp lý có \texttt{source\_id} tồn tại và khớp nội dung quy định.
    \item \textbf{Legal Temporal Correctness:} Tỷ lệ các văn bản làm căn cứ trả lời còn nguyên giá trị hiệu lực tại thời điểm tham chiếu \texttt{as\_of\_date}.
\end{itemize}

\section{Phân tích kết quả thực nghiệm và thảo luận khoa học}

Dựa trên số liệu thực nghiệm định lượng, nhóm nghiên cứu rút ra các kết luận khoa học quan trọng:
\begin{enumerate}
    \item \textbf{Khắc phục triệt để ảo giác (Giải quyết RQ1):} Traditional RAG ($S_1$) chỉ đạt Faithfulness $67.05\%$ do nạp cả những đoạn trích nhiễu vào bộ sinh. Trong khi đó, Full CRAG ($S_4$) nâng chỉ số Faithfulness lên tới \textbf{$92.86\% - 100.0\%$}, chứng minh cơ chế tinh lọc \textit{Knowledge Refinement} đã loại bỏ hoàn toàn các mệnh đề suy diễn không căn cứ.
    \item \textbf{Độ bao phủ ngữ cảnh toàn diện (Context Recall = 1.0):} $S_1$ và $S_2$ chỉ đạt Context Recall $66.67\%$ do bị giới hạn bởi Top-K cố định hoặc tìm kiếm web tràn lan thiếu cấu trúc. $S_4$ đạt Context Recall \textbf{$100\%$}, minh chứng cho hiệu quả kết hợp giữa truy hồi lai và bóc tách dải quy định pháp lý.
    \item \textbf{Tuân thủ hiệu lực pháp lý (Temporal Correctness = 1.0):} Các hệ thống $S_1, S_2$ thường xuyên trích dẫn văn bản cũ đã hết hiệu lực nếu câu từ của văn bản cũ khớp từ khóa tốt hơn văn bản mới. Nhờ sự hỗ trợ của \texttt{legal/citations.py} và \texttt{legal/temporal.py}, $S_4$ đạt độ chính xác thời gian pháp lý tuyệt đối ($100\%$).
    \item \textbf{Sự đánh đổi giữa độ chính xác và độ trễ:} Thời gian phản hồi của $S_4$ ($142.77$s) cao hơn đáng kể so với $S_1$ ($69.65$s) do phải thực thi chuỗi: truy hồi lai $\to$ RRF $\to$ Cross-Encoder $\to$ tinh lọc dải quy định $\to$ tìm kiếm web $\to$ kiểm định trích dẫn. Tuy nhiên, trong lĩnh vực pháp lý doanh nghiệp, \textbf{tính chính xác và tính chịu trách nhiệm pháp lý là ưu tiên sống còn, vượt trên tốc độ phản hồi đơn thuần}.
\end{enumerate}

\section{Phân tích trường hợp thất bại thực tế}

Để bảo đảm tính khách quan học thuật, nhóm tiến hành mổ xẻ một ca thất bại thực tế (\textit{failure case}) ghi nhận trong quá trình thử nghiệm:
\begin{itemize}
    \item \textbf{Câu hỏi thử nghiệm:} \textit{“Văn bản số 01/2021/TT-BKHĐT hướng dẫn đăng ký doanh nghiệp có bị bãi bỏ bởi Thông tư 02/2023/TT-BKHĐT hay không?”}
    \item \textbf{Hiện tượng ghi nhận:} Mặc dù cả hai Thông tư đều có trong kho dữ liệu, hệ thống $S_4$ chỉ truy hồi được Điều 1 của Thông tư 02 mà bỏ sót Điều khoản thi hành ở cuối văn bản (nơi ghi nhận danh mục văn bản bãi bỏ). Kết quả chấm điểm của bộ đánh giá chỉ đạt $0.58$ (nhánh \textbf{AMBIGUOUS}), kích hoạt tìm kiếm ngoài nhưng không bóc tách được bảng đối chiếu chi tiết.
    \item \textbf{Nguyên nhân gốc rễ:}
    \begin{enumerate}
        \item \textit{Đứt gãy liên kết dẫn chiếu chéo:} Mệnh đề bãi bỏ thường nằm ở chương cuối (\textit{Điều khoản thi hành}), cách rất xa phần định nghĩa nội dung nghiệp vụ. Vector embedding BGE-M3 của đoạn câu hỏi nghiệp vụ không có độ tương đồng cosine cao với một câu kỹ thuật hành chính: \textit{“Thông tư này bãi bỏ Thông tư số 01/2021/TT-BKHĐT...”}
        \item \textit{Giới hạn của Top-K Reranker:} Bộ xếp hạng lại chỉ giữ lại 5 ứng viên hàng đầu, dẫn đến việc đoạn chứa điều khoản thi hành bị trượt khỏi cửa sổ ngữ cảnh nạp vào bộ sinh.
    \end{enumerate}
    \item \textbf{Giải pháp khắc phục đã áp dụng:} Bổ sung module \texttt{legal/temporal.py} và bảng \texttt{legal\_relations} trong cơ sở dữ liệu SQLite. Khi câu hỏi chứa cặp số hiệu văn bản, hệ thống kích hoạt truy vấn quan hệ cấu trúc \texttt{SELECT relation\_type FROM legal\_relations} thay vì chỉ dựa vào truy hồi văn bản tự do, giải quyết dứt điểm các ca truy vấn hiệu lực lịch sử.
\end{itemize}
