\section{PHỤ LỤC C: Tiêu chí Tự Kiểm tra Chất lượng Trước khi Nộp}

\begin{calloutbox}{Bảng kiểm tra điều kiện nghiệm thu}
\begin{itemize}
    \item Có ít nhất một ca kiểm thử \textbf{Correct} trả lời chính xác và dẫn nguồn hợp lệ.
    \item Có ít nhất một ca kiểm thử \textbf{Incorrect} kích hoạt thành công tìm kiếm web có kiểm soát.
    \item Có ít nhất một ca kiểm thử \textbf{Ambiguous} hợp nhất thành công tri thức nội bộ và mở rộng.
    \item Có ít nhất một truy vấn metadata trực tiếp từ cơ sở dữ liệu quan hệ.
    \item Vết thực thi (\textit{Execution Trace}) thể hiện rõ điểm số đánh giá và nhánh rẽ được chọn.
    \item Bảng thực nghiệm so sánh đầy đủ giữa Traditional RAG, RAG + Always Web và Full CRAG.
    \item Tập Calibration Set được phân tách độc lập hoàn toàn khỏi Held-out Test Set.
    \item Có Citation Validator và chứng minh được khả năng chặn câu trả lời trích dẫn sai.
    \item Có phân tích ca thất bại thực tế kèm nguyên nhân gốc rễ.
    \item Tuyệt đối không dùng dữ liệu memory làm căn cứ pháp lý.
    \item Đạt chỉ số an toàn tuyệt đối: \textbf{Cross-client Contamination = 0}.
\end{itemize}
\end{calloutbox}
