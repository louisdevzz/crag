\textbf{Legal-aware Chunking:} Không dùng kỹ thuật cắt đoạn cố định (\textit{fixed-size recursive chunking 700/100}) gây đứt gãy câu chữ pháp lý. Thay vào đó, văn bản được phân tách theo đúng cấu trúc thứ bậc:
\[
\text{Văn bản} \;\longrightarrow\; \text{Chương} \;\longrightarrow\; \text{Điều} \;\longrightarrow\; \text{Khoản} \;\longrightarrow\; \text{Điểm}

\end{keypoint}

from FlagEmbedding import FlagReranker
from math import exp
from config import RERANKER_MODEL

reranker = FlagReranker(RERANKER_MODEL, use_fp16=True)

def sigmoid(x: float) -> float:
    return 1.0 / (1.0 + exp(-float(x)))

def score_pairs(query: str, docs: list[dict]) -> list[float]:
    pairs = [[query, d["text"]] for d in docs]
    raw_scores = reranker.compute_score(pairs)
    return [sigmoid(s) for s in raw_scores]

\section{BƯỚC 9: Xây dựng các Tools}

\subsection{Reciprocal Rank Fusion (RRF)}
Hợp nhất kết quả từ Dense và Lexical retriever theo công thức:
\begin{equation}
S_{\text{RRF}}(d) = \sum_{r \in \mathcal{R}} \frac{1}{k + \text{rank}_r(d)} \quad (\text{với } k = 60)
\end{equation}

\begin{lstlisting}[language=Python, caption={Thuật toán Reciprocal Rank Fusion}]
def rrf_merge(result_lists: list[list[dict]], k: int = 60) -> list[dict]:
    scores = {}
    items = {}
    for results in result_lists:
        for rank, item in enumerate(results, start=1):
            key = item["evidence_id"]
            scores[key] = scores.get(key, 0.0) + 1.0 / (k + rank)
            items[key] = item
    ordered_keys = sorted(scores, key=scores.get, reverse=True)
    return [items[k] for k in ordered_keys]
\end{lstlisting}

def decide_crag_action(scores: list[float]) -> str:
    best_score = max(scores) if scores else 0.0

    strips = []

\section{BƯỚC 15: Kiểm định Trích dẫn (Citation Validator)}

Hệ thống triển khai cơ chế kiểm tra trích dẫn xác định (\textit{Deterministic Citation Validation}):

\begin{lstlisting}[language=Python, caption={Hàm kiểm định trích dẫn xác định}]
def validate_citations(answer: dict, evidence_map: dict, as_of_date: str = None) -> dict:
    errors = []
    for claim in answer.get("claims", []):
        for sid in claim.get("source_ids", []):
            if sid not in evidence_map:
                errors.append(f"Lỗi: source_id không tồn tại trong tập căn cứ: {sid}")
                continue
            if as_of_date and not is_effective_at(evidence_map[sid], as_of_date):
                errors.append(f"Lỗi: Văn bản {sid} không còn hiệu lực tại ngày {as_of_date}")
    return {"ok": len(errors) == 0, "errors": errors}
\end{lstlisting}

def build_crag_graph():
    g = StateGraph(AgentState)
    
    # Dang ky cac node
    g.add_node("router", route_question)
    g.add_node("db", query_db)
    g.add_node("retrieve", hybrid_retrieve)
    g.add_node("evaluate", evaluate_retrieval)
    g.add_node("refine", refine_internal_node)
    g.add_node("rewrite", rewrite_query_node)
    g.add_node("web", web_search_node)
    g.add_node("select_web", select_external_node)
    g.add_node("merge", merge_evidence_node)
    g.add_node("generate", generate_answer)
    g.add_node("cite_validate", validate_citations_node)
    
    # Canh dieu kien sau evaluate: CORRECT -> refine, AMBIGUOUS -> refine, INCORRECT -> rewrite
    g.add_conditional_edges(
        "evaluate",
        lambda s: s["crag_action"],
        {"CORRECT": "refine", "AMBIGUOUS": "refine", "INCORRECT": "rewrite"}
    )
    
    # Dieu huong sau refine
    g.add_conditional_edges(
        "refine",
        lambda s: "generate" if s["crag_action"] == "CORRECT" else "rewrite",
        {"generate": "generate", "rewrite": "rewrite"}
    )
    
    # Nhanh ngoai
    g.add_edge("rewrite", "web")
    g.add_edge("web", "select_web")
    g.add_conditional_edges(
        "select_web",
        lambda s: "merge" if s["crag_action"] == "AMBIGUOUS" else "generate",
        {"merge": "merge", "generate": "generate"}
    )
    
    g.add_edge("merge", "generate")
    g.add_edge("generate", "cite_validate")
    g.add_edge("cite_validate", END)
    
    return g.compile()
\end{lstlisting}
