% --- Custom Callouts in Clean Academic Style ---
\setlength{\fboxsep}{8pt}
\setlength{\fboxrule}{0.8pt}

\newsavebox{\calloutboxsave}
\newenvironment{calloutbox}[2][]{%
  \begin{lrbox}{\calloutboxsave}%
  \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}%
    {\bfseries\small\color{tableheader} #2}\par\vspace{4pt}%
    \small\color{callouttext}%
}{%
  \end{minipage}%
  \end{lrbox}%
  \par\addvspace{6pt}\noindent
  \fcolorbox{calloutborder}{calloutbg}{\usebox{\calloutboxsave}}%
  \par\addvspace{6pt}%
}

\newsavebox{\infoboxsave}
\newenvironment{infobox}[2][]{%
  \begin{lrbox}{\infoboxsave}%
  \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}%
    {\bfseries\small\color{tableheader} #2}\par\vspace{4pt}%
    \small%
}{%
  \end{minipage}%
  \end{lrbox}%
  \par\addvspace{6pt}\noindent
  \fcolorbox{infoborder}{infobg}{\usebox{\infoboxsave}}%
  \par\addvspace{6pt}%
}

\newsavebox{\keypointsave}
\newenvironment{keypoint}[1][]{%
  \begin{lrbox}{\keypointsave}%
  \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}%
    \small%
}{%
  \end{minipage}%
  \end{lrbox}%
  \par\addvspace{6pt}\noindent
  \fcolorbox{black}{lightbg}{\usebox{\keypointsave}}%
  \par\addvspace{6pt}%
}
